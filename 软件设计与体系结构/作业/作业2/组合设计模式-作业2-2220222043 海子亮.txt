class Graphic {
public:
	virtual void draw = 0;
	virtual void add(Graphic* graphic) = 0;
	virtual void remove(Graphic* graphic) = 0;
	virtual Graphic* getChild(int index) = 0;
};

class Line :public Graphic {
public:
	void draw() override {
		// draw a line
		void add(Graphic * graphic)override {}
		void remove(Graphic * graphic)override {}
		Graphic* getChild(int index)override { return nullptr; }

	};
class Rectangle:public Graphic{
public:
	void draw()override {
		// draw a rectangle
		void add(Graphic * graphic)override {}
		void remove(Graphic * graphic)override {}
		Graphic* getChild(int index)override { return nullptr; }
	};

	class Text :public Graphic {
	public:
		void draw()override {
			// draw text
			void add(Graphic * graphic)override {}
			void remove(Graphic * graphic)override {}
			Graphic* getChild(int index)override { return nullptr; }
		};

		class Picture :public Graphic {
		public:
			void draw()override {
				// draw a picture
				for (auto graphic : children) {
					graphic->fraw();
				}
			}auto it = find(children.begin(), children.end(), graphic);
			if (it != children.end()) {
				children.erase(it);
			}
		};

		void add(Graphic* graphic) override {
			children.push_back(graphic);
		}
		Graphic* getChild(int index) override {
			if (index >= 0 && index < children.size()) {
				return children[index];
			}
			else {
				return nullptr;
			}
		}
	private:
		vector<Graphic*>children;
	};
